"""Mutation run over a SCRATCH COPY of the working tree (the real repository is never touched).

usage: mutate_all.py <repo> <scratch-copy-dir> <python>

For each mutant: edit one file in the copy, run the tests that are supposed to protect it, expect a
non-zero exit (KILLED).  A passing run is a SURVIVOR and is reported.  Every pattern must match exactly
once, so a stale mutant fails loudly instead of silently becoming a no-op.
"""

extract = subprocess.Popen(["tar", "-x", "-C", str(COPY)], stdin=tar.stdout)
assert tar.stdin is not None
tar.stdin.write(files)
tar.stdin.close()
extract.wait()
tar.wait()
assert (COPY / "scripts/ci_docker_preflight.py").exists()

only = sys.argv[4:]  # optional name prefixes
survivors: list[str] = []
killed_n = 0
for name, rel, old, new, tests in M:
    if only and not any(name.startswith(o) for o in only):
        continue
    f = COPY / rel
    orig = f.read_text()
    if orig.count(old) != 1:
        print("STALE   ", name, f"(pattern matched {orig.count(old)}x)")
        survivors.append(name + " [STALE PATTERN]")
        continue
    f.write_text(orig.replace(old, new))
    try:
        r = subprocess.run([PY, "-m", "pytest", "-p", "no:cacheprovider", "-q", "-x", "-o", "addopts=", *tests], cwd=COPY, capture_output=True, text=True, timeout=600)
    finally:
        f.write_text(orig)
    killed = r.returncode != 0
    killed_n += killed
    first = next((ln for ln in r.stdout.splitlines() if ln.startswith("FAILED")), "")
    print(("KILLED  " if killed else "SURVIVED"), name, ("| " + first.split("::")[-1][:90]) if first else "")
    if not killed:
        survivors.append(name)
print(f"killed {killed_n} / {len([1 for n, *_ in M if not only or any(n.startswith(o) for o in only)])}")
print("survivors:", survivors)
